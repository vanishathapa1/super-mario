let audioContext = null;

let muted = false;


function initAudio() {

    if (!audioContext) {

        try {

            audioContext = new (window.AudioContext ||

                window.webkitAudioContext)();

        } catch {

            audioContext = null;

        }

    }


    if (audioContext && audioContext.state === "suspended") {

        audioContext.resume();

    }

}


function beep(freq, duration = .08, type = "square", volume = .035) {


    if (muted || !audioContext) return;


    const osc = audioContext.createOscillator();

    const gain = audioContext.createGain();


    osc.type = type;

    osc.frequency.value = freq;


    gain.gain.setValueAtTime(volume, audioContext.currentTime);

    gain.gain.exponentialRampToValueAtTime(

        0.001,

        audioContext.currentTime + duration

    );


    osc.connect(gain);

    gain.connect(audioContext.destination);


    osc.start();

    osc.stop(audioContext.currentTime + duration);

}

const particles = [];


function spawnParticles(x, y, amount, options = {}) {


    for (let i = 0; i < amount; i++) {


        particles.push({

            x,

            y,

            vx: rand(-options.spread || -140, options.spread || 140),

            vy: rand(-options.power || -200, -30),

            life: rand(.35, .75),

            maxLife: .75,

            size: rand(3, 7),

            gravity: options.gravity || 500,

            type: options.type || "circle"

        });

    }

}


function updateParticles(dt) {


    for (let i = particles.length - 1; i >= 0; i--) {


        const p = particles[i];


        p.life -= dt;


        if (p.life <= 0) {

            particles.splice(i, 1);

            continue;

        }


        p.x += p.vx * dt;

        p.y += p.vy * dt;

        p.vy += p.gravity * dt;

    }

}


function drawParticles(cameraX) {


    for (const p of particles) {


        ctx.globalAlpha = clamp(p.life / p.maxLife, 0, 1);


        ctx.beginPath();

        ctx.arc(

            p.x - cameraX,

            p.y,

            p.size,

            0,

            Math.PI * 2

        );


        ctx.fillStyle = "#fff";

        ctx.fill();

    }


    ctx.globalAlpha = 1;

}

const player = {

    x: 150,

    y: 500,

    w: 36,

    h: 52,


    vx: 0,

    vy: 0,


    grounded: false,

    facing: 1,


    invulnerable: 0,

    shield: 0,


    anim: 0,

    checkpointX: 150,

    checkpointY: 500

};


function resetPlayer() {


    player.x = player.checkpointX;

    player.y = player.checkpointY;


    player.vx = 0;

    player.vy = 0;


    player.grounded = false;

    player.invulnerable = 1.6;

    player.shield = 0;

}


function updatePlayer(dt) {


    if (player.invulnerable > 0)

        player.invulnerable -= dt;


    if (player.shield > 0)

        player.shield -= dt;


    let acceleration = CONFIG.moveAcceleration;


    if (keys.left) {

        player.vx -= acceleration * dt;

        player.facing = -1;

    }


    if (keys.right) {

        player.vx += acceleration * dt;

        player.facing = 1;

    }


    if (!keys.left && !keys.right) {


        if (player.vx > 0) {

            player.vx -= CONFIG.friction * dt;

            if (player.vx < 0) player.vx = 0;

        }


        if (player.vx < 0) {

            player.vx += CONFIG.friction * dt;

            if (player.vx > 0) player.vx = 0;

        }

    }


    player.vx = clamp(

        player.vx,

        -CONFIG.maxSpeed,

        CONFIG.maxSpeed

    );


    if (jumpQueued && player.grounded) {


        player.vy = -CONFIG.jumpPower;


        if (powerJumpTimer > 0)

            player.vy *= 1.22;


        player.grounded = false;


        jumpSound();

        spawnParticles(

            player.x + player.w / 2,

            player.y + player.h,

            7,

            { spread: 80, power: 150 }

        );

    }


    jumpQueued = false;


    player.vy += CONFIG.gravity * dt;

    player.vy = clamp(player.vy, -1200, 1100);


    player.x += player.vx * dt;


    resolveHorizontalCollisions();


    player.y += player.vy * dt;


    player.grounded = false;


    resolveVerticalCollisions();


    player.x = clamp(

        player.x,

        0,

        CONFIG.worldWidth - player.w

    );


    if (player.y > CONFIG.levelHeight + 200) {

        loseLife();

    }


    player.anim += dt * (Math.abs(player.vx) / 50 + 2);

}

function resolveHorizontalCollisions() {


    for (const p of platforms) {


        if (!rectsOverlap(player, p))

            continue;


        if (player.vx > 0) {

            player.x = p.x - player.w;

            player.vx = 0;

        }


        else if (player.vx < 0) {

            player.x = p.x + p.w;

            player.vx = 0;

        }

    }

}


function resolveVerticalCollisions() {


    for (const p of platforms) {


        if (!rectsOverlap(player, p))

            continue;


        if (player.vy > 0) {


            const previousBottom =

                player.y + player.h - player.vy * deltaTime;


            if (previousBottom <= p.y + 12) {


                player.y = p.y - player.h;

                player.vy = 0;

                player.grounded = true;


                if (p.type === "bouncy") {

                    player.vy = -900;

                    player.grounded = false;

                    beep(500, .1);

                }

            }

        }


        else if (player.vy < 0) {


            player.y = p.y + p.h;

            player.vy = 50;

        }

    }

}

            ctx.beginPath();

            ctx.ellipse(-25, -4, 15, 8, -.3, 0, Math.PI * 2);

            ctx.fill();


            ctx.beginPath();

            ctx.ellipse(25, -4, 15, 8, .3, 0, Math.PI * 2);

            ctx.fill();

        }


        ctx.restore();

    }

}


/* =========================================================

   RENDER POWER UPS

========================================================= */


function drawPowerUps() {


    for (const p of powerUps) {


        if (p.collected) continue;


        const x = p.x - camera.x;

        const y = p.y + Math.sin(p.phase) * 5;


        ctx.save();


        ctx.translate(x + p.w / 2, y + p.h / 2);


        ctx.fillStyle =

            p.type === "shield" ?

            "#48d9ff" :

            "#a66cff";


        ctx.beginPath();

        ctx.arc(0, 0, 18, 0, Math.PI * 2);

        ctx.fill();


        ctx.fillStyle = "white";

        ctx.font = "bold 17px Arial";

        ctx.textAlign = "center";

        ctx.textBaseline = "middle";


        ctx.fillText(

            p.type === "shield" ? "S" : "J",

            0,

            1

        );


        ctx.restore();

    }

}


/* =========================================================

   CHECKPOINT & GOAL

========================================================= */


function drawCheckpoints() {


    for (const cp of checkpoints) {


        const x = cp.x - camera.x;


        ctx.fillStyle = "#49352d";


        ctx.fillRect(

            x,

            cp.y - 80,

            8,

            80

        );


        ctx.fillStyle =

            cp.active ?

            "#55e47b" :

            "#d8d8d8";


        ctx.beginPath();


        ctx.moveTo(x + 8, cp.y - 75);

        ctx.lineTo(x + 55, cp.y - 58);

        ctx.lineTo(x + 8, cp.y - 40);


        ctx.closePath();

        ctx.fill();

    }

}


function drawGoal() {


    const x = goal.x - camera.x;


    ctx.fillStyle = "#49352d";

    ctx.fillRect(x, goal.y, 8, 200);


    ctx.fillStyle = "#ffdc4a";


    ctx.beginPath();


    ctx.moveTo(x + 8, goal.y + 10);

    ctx.lineTo(x + 65, goal.y + 28);

    ctx.lineTo(x + 8, goal.y + 48);


    ctx.closePath();

    ctx.fill();


    ctx.fillStyle = "#ffffff";

    ctx.font = "bold 14px Arial";

    ctx.fillText("GO!", x + 17, goal.y + 38);

}

/* =========================================================

   TIMER

========================================================= */


let timerAccumulator = 0;


function updateTimer(dt) {


    timerAccumulator += dt;


    if (timerAccumulator >= 1) {


        const seconds = Math.floor(timerAccumulator);


        timeLeft -= seconds;

        timerAccumulator -= seconds;


        if (timeLeft <= 0) {


            timeLeft = 0;

            updateHUD();

            loseLife();


        } else {


            updateHUD();

        }

    }

}

    buildLevel();

    buildCoins();

    buildEnemies();

    buildPowerUps();


    checkpoints.forEach(cp => cp.active = false);


    player.checkpointX = 150;

    player.checkpointY = 620 - player.h;


    resetPlayer();


    camera.x = 0;


    timeLeft = CONFIG.levelTime;

    timerAccumulator = 0;


    updateHUD();

}


/* =========================================================

   GAME LOOP

========================================================= */


let lastTime = performance.now();

let deltaTime = 0;


function gameLoop(now) {


    deltaTime = Math.min(

        (now - lastTime) / 1000,

        .05

    );


    lastTime = now;


    if (gameState === "PLAYING") {


        updateTimer(deltaTime);

        updatePlatforms(deltaTime);

        updatePlayer(deltaTime);

        updateCoins(deltaTime);

        updateEnemies(deltaTime);

        updatePowerUps(deltaTime);

        updateCheckpoints();

        updateGoal();

        updateParticles(deltaTime);

        updateCamera(deltaTime);

    }


    render();


    requestAnimationFrame(gameLoop);

}


requestAnimationFrame(gameLoop);

    winSound();

        .textContent = formatScore(score - bonus);


    document.getElementById("timeBonus")

        .textContent = formatScore(bonus);


    document.getElementById("winCoins")

        .textContent = coins;


    document.getElementById("winTotal")

        .textContent = formatScore(score);

        bestLevel = levelNumber;

        localStorage.setItem(

            "pixelQuestBestLevel",

        );

        .classList.add("hidden");

        .classList.add("hidden");

        .classList.add("hidden");

}
